// 🔹 CLASSIFICADOR ASSIM
// =========================
function classificarAssim(status: string) {
  const s = (status || "").toLowerCase()

  if (s.includes("liberado") && !s.includes("*")) return "autorizado"
  if (s.includes("liberado") && s.includes("*")) return "estornado"
  if (/\d{3,}/.test(s)) return "pendencia_adm"

  return null
}

// 🔹 SINCRONIZA STATUS ASSIM → FILA

async function sincronizarStatusAssim(supabase: any) {
  console.log("🔄 Sincronizando status da ASSIM...")

  const { data: fila } = await supabase
    .from("fila_autorizacoes")
    .select("*")
    .is("status_assim", null)

  const { data: assim } = await supabase
    .from("autorizacoes_assim")
    .select("*")
    .gte(
      "data_execucao",
      new Date(Date.now() - 2 * 86400000).toISOString()
    )

  // 🔹 FUNÇÃO DE DIFERENÇA DE HORÁRIO
  function diffMinutos(h1: string, h2: string) {
    const [h1h, h1m] = h1.split(":").map(Number)
    const [h2h, h2m] = h2.split(":").map(Number)

    return Math.abs((h1h * 60 + h1m) - (h2h * 60 + h2m))
  }

	// ORDENAR FILA
	const filaOrdenada = [...(fila || [])].sort((a, b) => {
		return a.horario.localeCompare(b.horario)
	})

	// 🔥 CONTROLE DE USO
	const usados = new Set()

// 🔹 PROCESSA FILA
for (const f of filaOrdenada || []) {
  if (!f.paciente_id || !f.horario) continue

  const horaFila = String(f.horario).slice(0, 5)

  const candidatos: any[] = []

  for (const a of assim || []) {
    if (!a.data_execucao || !a.paciente_id) continue

    const d = new Date(a.data_execucao)
    const data = d.toISOString().slice(0, 10)
    const horaAssim = d.toTimeString().slice(0, 5)

    if (
      String(a.paciente_id) === String(f.paciente_id) &&
      data === f.data_atendimento &&
      a.codigo_tuss &&
      f.tuss &&
      String(a.codigo_tuss) === String(f.tuss)
    ) {
      const diff = diffMinutos(horaAssim, horaFila)
      const chaveUso = `${a.paciente_id}_${data}_${horaAssim}`

      if (diff <= 30 && !usados.has(chaveUso)) {
        candidatos.push({
          a,
          diff,
          chaveUso,
          status: classificarAssim(a.status)
        })
      }
    }
  }

  // 🔥 SE NÃO TEM CANDIDATO → pula
  if (candidatos.length === 0) continue

  // 🔥 PRIORIDADE DE STATUS
  function prioridade(status: string | null) {
    if (status === 'autorizado') return 1
    if (status === 'estornado') return 2
    if (status === 'pendencia_adm') return 3
    return 999
  }

  // 🔥 ORDENA: primeiro melhor status, depois menor diff
  candidatos.sort((x, y) => {
    const p1 = prioridade(x.status)
    const p2 = prioridade(y.status)

    if (p1 !== p2) return p1 - p2
    return x.diff - y.diff
  })

  const melhor = candidatos[0]
  const melhorMatch = melhor.a

  // 🔥 MARCA COMO USADO
  usados.add(melhor.chaveUso)

  const status = classificarAssim(melhorMatch.status)

  // 🔥 NÃO SOBRESCREVE GUIA REAL (mas permite MANUAL)
  if (
    f.status_assim === status &&
    f.numero_autorizacao &&
    f.numero_autorizacao !== "MANUAL"
  ) continue

  await supabase
    .from("fila_autorizacoes")
    .update({
      status_assim: status,
      numero_autorizacao: melhorMatch.guia,
      assim_updated_at: new Date().toISOString(),
    })
    .eq("id", f.id)
}

console.log("✅ Status sincronizados")
}

    // =========================

        try {
          const res = await fetch(url)

          if (!res.ok) return null

          const agendas = await res.json()

          if (!Array.isArray(agendas)) return null

          return agendas
            .filter((a: any) => {
              const terapia = (a.terapia || "")
                .trim()
                .toLowerCase()
                .normalize("NFD")
                .replace(/[\u0300-\u036f]/g, "")

              return ![
                "estagio",
                "facilitador tecnico",
                "operacoes clinicas",
                "aplicador suporte",
                "apoio operacional",
                "supervisao aba",
                "aplicador aba escola",
                "aplicador aba casa",
                "tecnico terapeutico particular",
                "avaliacao neuropsicologica",
              ].includes(terapia)
            })
            .map((a: any) => {
              const terapiaNormalizada = (a.terapia || "")
                .trim()
                .toLowerCase()
                .normalize("NFD")
                .replace(/[\u0300-\u036f]/g, "")

              return {
                paciente_id: p.id,
                paciente_nome: a.nome,
                empresa: a.empresa,
                matricula: a.matricula?.slice(0, 7) || null,
                dep: a.dep,
                crm: a.crm?.replace(/\D/g, ""),

                tuss:

                data_atendimento: a.data,
                horario: a.horario,
                terapia: a.terapia,
                ativo: true,
                updated_at: new Date().toISOString(),
              }
            })
        } catch {
          return null
        }
      })

      const resLote = await Promise.allSettled(promessas)

      const dados = resLote
        .filter((r: any) => r.status === "fulfilled")
        .map((r: any) => r.value)
        .flat()
        .filter(Boolean)

      resultados.push(...dados)

      console.log(`📦 Lote ${i} processado`)
    }

    if (resultados.length === 0) {
      return new Response("Sem dados")
    }

    // =========================
    // 🧹 PASSO 4 — REMOVER DUPLICADOS
    // =========================
    const mapa = new Map()

    for (const item of resultados) {
      const chave = `${item.paciente_id}_${item.data_atendimento}_${item.horario}_${item.terapia}`
      mapa.set(chave, item)
    }

    const unicos = Array.from(mapa.values())

    // =========================
    // 💾 PASSO 5 — UPSERT
    // =========================
    for (let i = 0; i < unicos.length; i += 100) {
      const lote = unicos.slice(i, i + 100)

      const { error } = await supabase
        .from("agenda_orbita")
        .upsert(lote, {
          onConflict: "paciente_id,data_atendimento,horario,terapia",
        })

      if (error) throw error
    }

    // =========================
    // 🔥 PASSO FINAL — SINCRONIZA STATUS ASSIM
    // =========================
    await sincronizarStatusAssim(supabase)

    const tempo = Date.now() - inicio

    console.log(`✅ FINALIZADO em ${tempo}ms`)

    return new Response(
      JSON.stringify({
        status: "ok",
        registros: unicos.length,
        tempo_ms: tempo,
      }),
      { headers: { "Content-Type": "application/json" } }
    )
  } catch (err: any) {
    console.error("❌ ERRO:", err.message)

    return new Response(
      JSON.stringify({
        status: "erro",
        mensagem: err.message,
      }),
      { status: 500 }
    )
  }

import { serve } from "https://deno.land/std@0.168.0/http/server.ts";

const corsHeaders = {
  'Access-Control-Allow-Origin': '*',
  'Access-Control-Allow-Headers': 'authorization, x-client-info, apikey, content-type',
};

const CONCURRENCY = 5;

function hoje() {
  return new Date().toISOString().split("T")[0];
}

async function delay(ms: number) {
  return new Promise(res => setTimeout(res, ms));
}

async function fetchComRetry(url: string, options: any = {}, tentativas = 3) {
  for (let i = 0; i < tentativas; i++) {
    try {
      const res = await fetch(url, options);
      if (!res.ok) throw new Error(`Erro HTTP ${res.status}`);
      return await res.json();
    } catch (err) {
      console.log(`❌ Retry ${i + 1} falhou: ${url}`);
      await delay(1000 * (i + 1));
    }
  }
  return null;
}

async function buscarAtendimento(paciente: any, baseUrl: string, token: string) {
  const url = `${baseUrl}/agenda/detalhe?paciente_id=${paciente.paciente_id}&data=${hoje()}&horario=${paciente.horario}`;

  return await fetchComRetry(url, {
    method: "GET",
    headers: {
      Authorization: `Bearer ${token}`
    }
  });
}

async function processarPacientes(pacientes: any[], baseUrl: string, token: string) {
  const resultados: any[] = [];

  for (let i = 0; i < pacientes.length; i += CONCURRENCY) {
    const lote = pacientes.slice(i, i + CONCURRENCY);

    const respostas = await Promise.all(
      lote.map(p => buscarAtendimento(p, baseUrl, token))
    );

    resultados.push(...respostas.filter(r => r !== null));

    console.log(`📦 Processados ${Math.min(i + CONCURRENCY, pacientes.length)}/${pacientes.length}`);

    await delay(500);
  }

  return resultados;
}

serve(async (req) => {
  if (req.method === 'OPTIONS') {
    return new Response('ok', { headers: corsHeaders });
  }

  try {
    console.log("🚀 INICIOU SINCRONIZAÇÃO", new Date().toISOString());

    const baseUrl = Deno.env.get('ORBITA_BASE_URL');
    const supabaseUrl = Deno.env.get('SUPABASE_URL');
    const serviceKey = Deno.env.get('SUPABASE_SERVICE_ROLE_KEY');

    if (!baseUrl || !token || !supabaseUrl || !serviceKey) {
      throw new Error("Variáveis de ambiente faltando");
    }

    // 🔥 1. Buscar pacientes
    const pacientesResp = await fetchComRetry(
      `${baseUrl}/pacientes`,
    );

    if (!pacientesResp) throw new Error("Erro ao buscar pacientes");

    const pacientes = pacientesResp.data || pacientesResp || [];

    if (!Array.isArray(pacientes)) throw new Error("Formato inválido");

    console.log(`👥 Pacientes recebidos: ${pacientes.length}`);

    // 🔥 2. Buscar atendimentos em lote
    const atendimentos = await processarPacientes(pacientes, baseUrl, token);

    if (atendimentos.length === 0) {
      console.log("⚠️ Nenhum atendimento encontrado");
      return new Response("Sem dados", { headers: corsHeaders });
    }

    console.log(`✅ Atendimentos válidos: ${atendimentos.length}`);

    // 🔥 3. Normalizar
    const registros = atendimentos.map((a: any) => ({
      paciente_id: a.paciente_id || a.id,
      nome: a.paciente_nome || a.nome,
      data: hoje(),
      hora: a.horario_atendimento || a.horario,
      updated_at: new Date()
    }));

    // 🔥 4. UPSERT no Supabase
    const res = await fetch(`${supabaseUrl}/rest/v1/agenda_orbita`, {
      method: "POST",
      headers: {
        apikey: serviceKey,
        Authorization: `Bearer ${serviceKey}`,
        "Content-Type": "application/json",
        Prefer: "resolution=merge-duplicates"
      },
      body: JSON.stringify(registros)
    });

    if (!res.ok) throw new Error("Erro ao salvar no banco");

    console.log("🎉 FINALIZADO COM SUCESSO");

    return new Response("OK", { headers: corsHeaders });

  } catch (err) {
    console.error("💥 ERRO GERAL:", err.message);

    return new Response(JSON.stringify({ error: err.message }), {
      status: 500,
      headers: { ...corsHeaders, 'Content-Type': 'application/json' },
    });
  }
});

// Função Edge do Supabase para listar pacientes do Órbita
// Aceita apenas GET, inclui CORS, usa variáveis de ambiente

// Headers para CORS
const corsHeaders = {
  'Access-Control-Allow-Origin': '*',
  'Access-Control-Allow-Headers': 'authorization, x-client-info, apikey, content-type',
};

// Handler principal usando Deno.serve
Deno.serve(async (req) => {
  // Trata requisições OPTIONS para CORS
  if (req.method === 'OPTIONS') {
    return new Response('ok', { headers: corsHeaders });
  }

  // Permite apenas método GET
  if (req.method !== 'GET') {
    return new Response(JSON.stringify({ error: 'Método não permitido' }), {
      status: 405,
      headers: corsHeaders,
    });
  }

  // Obtém variáveis de ambiente
  const baseUrl = Deno.env.get('ORBITA_BASE_URL');
  const token = Deno.env.get('ORBITA_TOKEN');

  // Verifica se as variáveis estão definidas
  if (!baseUrl || !token) {
    return new Response(JSON.stringify({ error: 'Configuração faltando' }), {
      status: 500,
      headers: corsHeaders,
    });
  }

  try {
    // Faz a requisição GET para o endpoint do Órbita
    const response = await fetch(`${baseUrl}/pacientes`, {
      headers: {
        'Authorization': `Bearer ${token}`,
      },
    });

    // Verifica se a resposta foi bem-sucedida
    if (!response.ok) {
      return new Response(JSON.stringify({ error: 'Erro ao buscar pacientes' }), {
        status: response.status,
        headers: corsHeaders,
      });
    }

    // Parse do JSON da resposta
    const data = await response.json();

    // Extrai o array de pacientes, tentando diferentes estruturas
    let pacientesRaw = data;
    if (data.data && Array.isArray(data.data)) pacientesRaw = data.data;
    else if (data.items && Array.isArray(data.items)) pacientesRaw = data.items;
    else if (data.results && Array.isArray(data.results)) pacientesRaw = data.results;
    else if (data.pacientes && Array.isArray(data.pacientes)) pacientesRaw = data.pacientes;

    // Verifica se é um array válido
    if (!Array.isArray(pacientesRaw)) {
      return new Response(JSON.stringify({ error: 'Formato de resposta inválido' }), {
        status: 500,
        headers: corsHeaders,
      });
    }

    // Normaliza os dados dos pacientes com fallbacks
    const pacientes = pacientesRaw.map((p) => {
      const paciente_id = p.id || p.paciente_id || p.codigo || p.codigo_paciente;
      const paciente_nome = p.nome || p.paciente_nome || p.nome_paciente;
      const matricula = p.matricula || p.registro;
      return { paciente_id, paciente_nome, matricula, raw: p };
    });

    // Retorna a resposta normalizada
    return new Response(JSON.stringify({ pacientes }), {
      headers: corsHeaders,
    });
  } catch (error) {
    // Trata erros internos
    return new Response(JSON.stringify({ error: 'Erro interno' }), {
      status: 500,
      headers: corsHeaders,
    });
  }
});

import { serve } from "https://deno.land/std@0.168.0/http/server.ts";

serve(async (req) => {
  // Configurar headers CORS
  const corsHeaders = {
    'Access-Control-Allow-Origin': '*',
    'Access-Control-Allow-Headers': 'authorization, x-client-info, apikey, content-type',
    'Access-Control-Allow-Methods': 'POST',
  };

  // Responder a requisições OPTIONS para CORS
  if (req.method === 'OPTIONS') {
    return new Response('ok', { headers: corsHeaders });
  }

  // Aceitar apenas método POST
  if (req.method !== 'POST') {
    return new Response(JSON.stringify({ error: 'Método não permitido' }), {
      status: 405,
      headers: { ...corsHeaders, 'Content-Type': 'application/json' },
    });
  }

  try {
    // Obter variáveis de ambiente
    const orbitaBaseUrl = Deno.env.get('ORBITA_BASE_URL');
    const orbitaToken = Deno.env.get('ORBITA_TOKEN');
    if (!orbitaBaseUrl || !orbitaToken) {
      return new Response(JSON.stringify({ error: 'Configuração de ambiente faltando' }), {
        status: 500,
        headers: { ...corsHeaders, 'Content-Type': 'application/json' },
      });
    }

    // Parse do body JSON
    const body = await req.json();
    const { paciente_id, data, horario } = body;

    // Validar campos obrigatórios
    if (!paciente_id || !data || !horario) {
      return new Response(JSON.stringify({ error: 'Campos obrigatórios: paciente_id, data, horario' }), {
        status: 400,
        headers: { ...corsHeaders, 'Content-Type': 'application/json' },
      });
    }

    // Validar formato da data (YYYY-MM-DD)
    const dateRegex = /^\d{4}-\d{2}-\d{2}$/;
    if (!dateRegex.test(data)) {
      return new Response(JSON.stringify({ error: 'Formato de data inválido. Use YYYY-MM-DD' }), {
        status: 400,
        headers: { ...corsHeaders, 'Content-Type': 'application/json' },
      });
    }

    // Validar formato do horário (HH:mm ou HH:mm:ss)
    const timeRegex = /^\d{2}:\d{2}(:\d{2})?$/;
    if (!timeRegex.test(horario)) {
      return new Response(JSON.stringify({ error: 'Formato de horário inválido. Use HH:mm ou HH:mm:ss' }), {
        status: 400,
        headers: { ...corsHeaders, 'Content-Type': 'application/json' },
      });
    }

    // Construir URL da API
    const url = `${orbitaBaseUrl}/agenda/detalhe?paciente_id=${encodeURIComponent(paciente_id)}&data=${encodeURIComponent(data)}&horario=${encodeURIComponent(horario)}`;

    // Fazer requisição GET para a API externa
    const response = await fetch(url, {
      method: 'GET',
      headers: {
        'Authorization': `Bearer ${orbitaToken}`,
        'Content-Type': 'application/json',
      },
    });

    if (!response.ok) {
      return new Response(JSON.stringify({ error: 'Erro na API externa', status: response.status }), {
        status: response.status,
        headers: { ...corsHeaders, 'Content-Type': 'application/json' },
      });
    }

    // Obter resposta da API
    const rawResponse = await response.json();

    // Normalizar campos com fallbacks
    const normalizado = {
      paciente_id: rawResponse.paciente_id || rawResponse.id || rawResponse.codigo_paciente,
      data_atendimento: data,
      horario_atendimento: horario,
      data_horario: `${data}T${horario}${horario.length === 5 ? ':00' : ''}`, // Adicionar segundos se necessário
      paciente_nome: rawResponse.paciente_nome || rawResponse.nome_paciente || rawResponse.nome,
      empresa: rawResponse.empresa,
      matricula: rawResponse.matricula || rawResponse.registro,
      dep: rawResponse.dep || rawResponse.dependente,
      crm: rawResponse.crm,
      nome_medico: rawResponse.nome_medico || rawResponse.medico,
      terapia: rawResponse.terapia || rawResponse.procedimento,
      tuss1: rawResponse.tuss1 || rawResponse.tuss || rawResponse.codigo_tuss,
      orbita_agenda_id: rawResponse.orbita_agenda_id || rawResponse.agenda_id || rawResponse.id,
    };

    // Retornar resposta
    return new Response(JSON.stringify({
      atendimento: rawResponse,
      normalizado,
    }), {
      status: 200,
      headers: { ...corsHeaders, 'Content-Type': 'application/json' },
    });

  } catch (error) {
    // Capturar erros gerais
    return new Response(JSON.stringify({ error: 'Erro interno do servidor', details: error.message }), {
      status: 500,
      headers: { ...corsHeaders, 'Content-Type': 'application/json' },
    });
  }
});
