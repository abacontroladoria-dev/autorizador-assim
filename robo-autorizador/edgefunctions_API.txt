// _shared/utils.ts
interface CorsHeaders {
  'Access-Control-Allow-Origin': string;
  'Access-Control-Allow-Methods': string;
  'Access-Control-Allow-Headers': string;
}

const corsHeaders: CorsHeaders = {
  'Access-Control-Allow-Origin': '*',
  'Access-Control-Allow-Methods': 'GET, POST, OPTIONS',
  'Access-Control-Allow-Headers': 'authorization, x-client-info, apikey, content-type',
};

function handleCors(request: Request): Response | null {
  if (request.method === 'OPTIONS') {
    return new Response('ok', { headers: corsHeaders });
  }
  return null;
}

function jsonResponse(data: any, status = 200): Response {
  return new Response(JSON.stringify(data), {
    status,
    headers: { ...corsHeaders, 'Content-Type': 'application/json' },
  });
}

export { corsHeaders, handleCors, jsonResponse };

// supabase/functions/orbita-listar-pacientes/index.ts
import { handleCors, jsonResponse } from '../_shared/utils.ts';

Deno.serve(async (req: Request) => {
  // Verificar CORS
  const corsResponse = handleCors(req);
  if (corsResponse) return corsResponse;

  // Validar método HTTP
  if (req.method !== 'GET') {
    return jsonResponse({ error: 'Método não permitido' }, 405);
  }

  // Obter variáveis de ambiente
  const orbitaBaseUrl = Deno.env.get('ORBITA_BASE_URL');
  const orbitaToken = Deno.env.get('ORBITA_TOKEN');
  if (!orbitaBaseUrl || !orbitaToken) {
    return jsonResponse({ error: 'Configuração de ambiente faltando' }, 500);
  }

  try {
    // Chamar API do Órbita
    const response = await fetch(`${orbitaBaseUrl}/pacientes`, {
      method: 'GET',
      headers: {
        'Authorization': `Bearer ${orbitaToken}`,
        'Content-Type': 'application/json',
      },
    });

    if (!response.ok) {
      return jsonResponse({ error: 'Erro na API do Órbita', status: response.status }, response.status);
    }

    const data = await response.json();

    // Normalizar resposta
    // TODO: Ajustar mapeamento baseado no contrato final da API
    const pacientes = data.map((item: any) => ({
      paciente_id: item.id || item.paciente_id || item.pacienteId,
      paciente_nome: item.nome || item.paciente_nome || item.pacienteNome,
      matricula: item.matricula || item.registro,
      raw: item,
    }));

    return jsonResponse({ pacientes });
  } catch (error) {
    console.error('Erro ao chamar API:', error);
    return jsonResponse({ error: 'Erro interno do servidor' }, 500);
  }
});

// supabase/functions/orbita-consultar-atendimento/index.ts
import { handleCors, jsonResponse } from '../_shared/utils.ts';

Deno.serve(async (req: Request) => {
  // Verificar CORS
  const corsResponse = handleCors(req);
  if (corsResponse) return corsResponse;

  // Validar método HTTP
  if (req.method !== 'POST') {
    return jsonResponse({ error: 'Método não permitido' }, 405);
  }

  // Obter variáveis de ambiente
  const orbitaBaseUrl = Deno.env.get('ORBITA_BASE_URL');
  const orbitaToken = Deno.env.get('ORBITA_TOKEN');
  if (!orbitaBaseUrl || !orbitaToken) {
    return jsonResponse({ error: 'Configuração de ambiente faltando' }, 500);
  }

  try {
    // Ler body JSON
    const body = await req.json();
    const { paciente_id, data, horario } = body;

    // Validar parâmetros obrigatórios
    if (!paciente_id || !data || !horario) {
      return jsonResponse({ error: 'Parâmetros obrigatórios faltando: paciente_id, data, horario' }, 400);
    }

    // Construir URL com query params
    const url = `${orbitaBaseUrl}/agenda/detalhe?paciente_id=${encodeURIComponent(paciente_id)}&data=${encodeURIComponent(data)}&horario=${encodeURIComponent(horario)}`;

    // Chamar API do Órbita
    const response = await fetch(url, {
      method: 'GET',
      headers: {
        'Authorization': `Bearer ${orbitaToken}`,
        'Content-Type': 'application/json',
      },
    });

    if (!response.ok) {
      return jsonResponse({ error: 'Erro na API do Órbita', status: response.status }, response.status);
    }

    const rawResponse = await response.json();

    // Normalizar resposta
    // TODO: Ajustar baseado no contrato final da API
    return jsonResponse({ atendimento: rawResponse });
  } catch (error) {
    console.error('Erro ao chamar API:', error);
    return jsonResponse({ error: 'Erro interno do servidor' }, 500);
  }
});
